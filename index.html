ตัวอย่างโค้ดภาษา Python 

import requests
import subprocess

def get_hwid():
    # ดึงค่า UUID ของเครื่อง Windows
    cmd = 'wmic csproduct get uuid'
    return subprocess.check_output(cmd, shell=True).decode().split('\n')[1].strip()

def verify_key(license_key):
    api_url = "http://localhost:8080/api/license/verify"
    payload = {
        "key": license_key,
        "hwid": get_hwid()
    }
    
    response = requests.post(api_url, json=payload)
    data = response.json()
    
    if data.get("valid"):
        print(f"[+] เข้าสู่ระบบสำเร็จ! หมดอายุ: {data['data']['expiresAt']}")
        return True
    else:
        print(f"[-] ข้อผิดพลาด: {data.get('message')}")
        return False

# เรียกใช้งาน
verify_key("DEKTHEP-1D-A8F2-99C1-71B0")

ตัวอย่างโค้ดภาษา C# (.NET) 

using System;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;

class Program {
    static async Task Main() {
        var client = new HttpClient();
        var json = "{\"key\":\"DEKTHEP-1D-A8F2-99C1-71B0\",\"hwid\":\"PC-USER-001\"}";
        var content = new StringContent(json, Encoding.UTF8, "application/json");
        
        var response = await client.PostAsync("http://localhost:8080/api/license/verify", content);
        var result = await response.Content.ReadAsStringAsync();
        Console.WriteLine(result);
    }
}
